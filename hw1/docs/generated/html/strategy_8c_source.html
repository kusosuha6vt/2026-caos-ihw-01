45static uint64_t bounded(uint64_t *state, uint64_t bound) {
46 uint64_t threshold = (UINT64_C(0) - bound) % bound;

102int choose_painting(const Simulation *simulation, Visitor *visitor) {
103 uint64_t count = simulation->config->paintings;
104 if (simulation->config->strategy == CYCLIC)
105 return (int)((visitor->start + visitor->viewed * visitor->step) %
106 count);
107 uint64_t rank = simulation->config->strategy == RANDOM
108 ? bounded(&visitor->rng, count - visitor->viewed)
109 : 0;
110 int best = -1;
111 uint64_t smallest = UINT64_MAX;
112 for (uint64_t p = 0; p < count; ++p) {
113 if (visitor->seen[p])
114 continue;
115 if (simulation->config->strategy == LEAST_CROWDED) {
116 uint64_t crowd = simulation->paintings[p].viewers +
117 simulation->paintings[p].queue.length;
118 if (crowd < smallest) {
119 smallest = crowd;
120 best = (int)p;
121 }
122 } else if (!rank--)
123 return (int)p;
124 }
125 return best;
126}

131void strategy_init(const Config *config, Visitor *visitor) {
132 visitor->rng = mix(config->seed ^ mix(visitor->id) ^ UINT64_C(0x33cf));
133 visitor->start = bounded(&visitor->rng, config->paintings);
134 visitor->step = 0;
135 if (config->paintings > 1) {
136 visitor->step = 1 + bounded(&visitor->rng, config->paintings - 1);
137 while (gcd(visitor->step, config->paintings) != 1)
138 visitor->step = visitor->step % (config->paintings - 1) + 1;
139 }
