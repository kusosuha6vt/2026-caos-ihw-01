21static int check_queue(Simulation *s, Queue *q, State state, int painting) {
22 uint64_t length = 0;
23 int last = -1;
24 for (int i = q->head; i >= 0; i = s->visitors[i].next) {
25 if (i >= s->slots || ++length > (uint64_t)s->slots || s->membership[i])
26 return 0;
27 Visitor *v = &s->visitors[i];
28 if (v->state != state ||
29 (state == PAINTING_WAIT && v->painting != painting))
30 return 0;
31 s->membership[i] = 1;
32 last = i;
33 }
34 return length == q->length && last == q->tail && ((q->head < 0) == !length);
35}

40int simulation_check(Simulation *simulation) {
41 memset(simulation->membership, 0, (size_t)simulation->slots);
42 if (!check_queue(simulation, &simulation->entrance, ENTRANCE, -1))
43 return 0;
44 uint64_t viewing[GALLERY_MAX_PAINTINGS] = {0}, inside = 0, active = 0;
45 for (uint64_t p = 0; p < simulation->config->paintings; ++p)
46 if (!check_queue(simulation, &simulation->paintings[p].queue,
47 PAINTING_WAIT, (int)p))
48 return 0;
49 for (int i = 0; i < simulation->slots; ++i) {
50 Visitor *v = &simulation->visitors[i];
51 if (v->state == FREE)
52 continue;
53 ++active;
54 if (v->state != ENTRANCE)
55 ++inside;
56 if ((v->state == ENTRANCE || v->state == PAINTING_WAIT) !=
57 !!simulation->membership[i])
58 return 0;
59 uint64_t seen = 0;
60 for (uint64_t p = 0; p < simulation->config->paintings; ++p)
61 seen += v->seen[p];
62 if (seen != v->viewed || seen > simulation->config->paintings)
63 return 0;
64 if (v->state == VIEWING || v->state == PAINTING_WAIT) {
65 if (v->painting < 0 ||
66 (uint64_t)v->painting >= simulation->config->paintings ||
67 v->seen[v->painting])
68 return 0;
69 if (v->state == VIEWING) {
70 if (v->finish < simulation->now)
71 return 0;
72 ++viewing[v->painting];
73 }
74 }
75 }
76 for (uint64_t p = 0; p < simulation->config->paintings; ++p)
77 if (viewing[p] != simulation->paintings[p].viewers ||
78 viewing[p] > simulation->config->painting_capacity)
79 return 0;
80 return inside == simulation->gallery &&
81 inside <= simulation->config->capacity &&
82 active == simulation->active &&
83 simulation->arrived ==
84 simulation->completed + simulation->cancelled + active;
85}
