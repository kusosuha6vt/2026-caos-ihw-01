34static void push(Simulation *s, Queue *queue, int slot) {
35 s->visitors[slot].next = -1;
36 if (queue->tail >= 0)
37 s->visitors[queue->tail].next = slot;
38 else
39 queue->head = slot;
40 queue->tail = slot;

50static int pop(Simulation *s, Queue *queue) {
51 int slot = queue->head;
52 queue->head = s->visitors[slot].next;
53 if (queue->head < 0)
54 queue->tail = -1;
55 s->visitors[slot].next = -1;
56 --queue->length;
57 return slot;
58}

81static void start_waiters(Simulation *s, int painting) {
82 Painting *p = &s->paintings[painting];
83 while (p->queue.length && p->viewers < s->config->painting_capacity &&
84 !stop_signal && !s->io_failed) {
85 int slot = pop(s, &p->queue);
86 Visitor *v = &s->visitors[slot];
87 uint64_t wait = s->now - v->wait_since;
88 uint64_t duration = viewing_duration(s->config, v->id, painting);
89 v->state = VIEWING;
90 v->finish = s->now + duration;
91 ++p->viewers;
92 if (p->viewers > p->peak)
93 p->peak = p->viewers;
94 ++p->started;
95 ++s->views_started;
96 s->painting_wait_sum += wait;
97 if (wait > s->painting_wait_max)
98 s->painting_wait_max = wait;
99 output_event(s, "view_start",
100 "Начало осмотра; место занято до окончания", slot,
101 painting, duration);
102 }

111static void choose_or_leave(Simulation *s, int slot) {
112 Visitor *v = &s->visitors[slot];
113 if (v->viewed == s->config->paintings) {
114 --s->gallery;
115 --s->active;
116 ++s->completed;
117 output_event(s, "exit",
118 "Все картины осмотрены; место в галерее освобождено", slot,
119 -1, 0);
120 v->state = FREE;
121 return;
122 }

124 return;

126 v->painting = painting;

129 v->state = PAINTING_WAIT;
130 v->wait_since = s->now;
131 push(s, &s->paintings[painting].queue, slot);
132 if (s->paintings[painting].queue.head != slot ||
133 s->paintings[painting].viewers == s->config->painting_capacity)
134 output_event(s, "painting_wait",
135 "Ожидание свободного места в FIFO картины", slot, painting,
136 0);
137 start_waiters(s, painting);

146static void admit(Simulation *s) {
147 while (s->entrance.length && s->gallery < s->config->capacity &&
148 !stop_signal && !s->io_failed) {
149 int slot = pop(s, &s->entrance);
150 Visitor *v = &s->visitors[slot];
151 output_event(s, "permission", "Вахтёр разрешил вход первому в очереди",
152 slot, -1, 0);
153 uint64_t wait = s->now - v->wait_since;
154 s->entrance_wait_sum += wait;
155 if (wait > s->entrance_wait_max)
156 s->entrance_wait_max = wait;
157 ++s->entered;
158 ++s->gallery;
159 if (s->gallery > s->peak_gallery)
160 s->peak_gallery = s->gallery;
161 v->state = READY;

173static int arrive(Simulation *s) {
174 int slot;
175 for (slot = 0; slot < s->slots; ++slot)
176 if (s->visitors[slot].state == FREE)
177 break;
178 if (slot == s->slots) {
179 fprintf(
180 stderr,
181 "Исчерпан --max-active; увеличьте лимит или интервал прибытия.\n");
182 output_event(s, "limit",
183 "Достигнут лимит активных записей; день прерван с ошибкой",
184 -1, -1, 0);
185 return -1;
186 }
187 Visitor *v = &s->visitors[slot];
188 unsigned char *seen = v->seen;
189 *v = (Visitor){.state = ENTRANCE,
190 .id = ++s->arrived,
191 .wait_since = s->now,
192 .painting = -1,
193 .next = -1,
194 .seen = seen};
195 memset(v->seen, 0, (size_t)s->config->paintings);
196 strategy_init(s->config, v);
197 ++s->active;
198 push(s, &s->entrance, slot);
199 output_event(s, "arrival", "Посетитель прибыл к входу", slot, -1, 0);
200 if (s->gallery == s->config->capacity || s->entrance.length > 1)
201 output_event(s, "entrance_wait",
202 "Галерея заполнена; ожидание разрешения вахтёра", slot, -1,
203 0);
204 admit(s);
205 return 0;
206}

216 Visitor *v = &s->visitors[slot];

218 --s->paintings[painting].viewers;

239static void remove_waiter(Simulation *s, Queue *q, int slot) {
240 int previous = -1;
241 for (int i = q->head; i != slot; i = s->visitors[i].next)
242 previous = i;
243 if (previous >= 0)
244 s->visitors[previous].next = s->visitors[slot].next;
245 else
246 q->head = s->visitors[slot].next;
247 if (q->tail == slot)
248 q->tail = previous;
249 s->visitors[slot].next = -1;
250 --q->length;
251}

258static void cancel_active(Simulation *s) {
259 for (int i = 0; i < s->slots; ++i) {
260 Visitor *v = &s->visitors[i];
261 if (v->state == FREE)
262 continue;
263 int painting = (v->state == VIEWING || v->state == PAINTING_WAIT)
264 ? v->painting
265 : -1;
266 if (v->state == ENTRANCE)
267 remove_waiter(s, &s->entrance, i);
268 if (v->state == PAINTING_WAIT)
269 remove_waiter(s, &s->paintings[painting].queue, i);
270 if (v->state == VIEWING)
271 --s->paintings[painting].viewers;
272 if (v->state != ENTRANCE)
273 --s->gallery;
274 --s->active;
275 ++s->cancelled;
276 output_event(s, "cancel",
277 "Посещение прервано; занятые места освобождены", i,
278 painting, 0);
279 v->state = FREE;
280 }

290static int next_completion(const Simulation *s, uint64_t *time) {
291 uint64_t earliest = UINT64_MAX;
292 int slot = -1;
293 for (int i = 0; i < s->slots; ++i) {
294 const Visitor *v = &s->visitors[i];
295 if (v->state == VIEWING &&
296 (v->finish < earliest ||
297 (v->finish == earliest &&
298 (slot < 0 || v->id < s->visitors[slot].id)))) {
299 earliest = v->finish;
300 slot = i;
301 }
302 }
303 *time = earliest;
304 return slot;
305}

312int simulate(const Config *config) {
313 Simulation s = {.config = config, .log_fd = -1, .entrance = empty_queue()};
314 s.slots = (int)(config->unlimited || config->visitors > config->max_active
315 ? config->max_active
316 : config->visitors);
317 if (s.slots == 0)
318 s.slots = 1;
319 s.visitors = calloc((size_t)s.slots, sizeof(*s.visitors));
320 s.membership = calloc((size_t)s.slots, 1);
321 s.paintings = calloc((size_t)config->paintings, sizeof(*s.paintings));
322 int result = 1;
323 if (!s.visitors || !s.membership || !s.paintings) {
324 perror("calloc");
325 goto cleanup;
326 }
327 for (int i = 0; i < s.slots; ++i) {
328 s.visitors[i].seen = calloc((size_t)config->paintings, 1);
329 if (!s.visitors[i].seen) {
330 perror("calloc visitor");
331 goto cleanup;
332 }
333 }
334 for (uint64_t p = 0; p < config->paintings; ++p)
335 s.paintings[p].queue = empty_queue();
336 if (output_open(&s))
337 goto cleanup;
338 output_config(&s);
339 uint64_t next_arrival = config->unlimited || config->visitors
340 ? arrival_interval(config, 1)
341 : UINT64_MAX;
342 int failed = 0;
343 if (next_arrival == UINT64_MAX)
344 output_event(&s, "close", "Новые посетители больше не прибывают", -1,
345 -1, 0);
346 while (!stop_signal && !s.io_failed &&
347 (next_arrival != UINT64_MAX || s.active)) {
348 uint64_t earliest;
349 int slot = next_completion(&s, &earliest);
350 /* Tie rule: completion precedes arrival, then smaller visitor ID. */
351 if (slot >= 0 && earliest <= next_arrival) {
352 advance(&s, earliest);
353 finish_view(&s, slot);
354 } else if (next_arrival != UINT64_MAX) {
355 advance(&s, next_arrival);
356 if (arrive(&s)) {
357 failed = 1;
358 break;
359 }
360 if (!config->unlimited && s.arrived == config->visitors) {
361 next_arrival = UINT64_MAX;
362 output_event(&s, "close",
363 "Новые прибытия закрыты; очередь будет обслужена",
364 -1, -1, 0);
365 } else {
366 uint64_t interval = arrival_interval(config, s.arrived + 1);
367 if (s.now >= UINT64_MAX - interval - config->view_max ||
368 s.arrived == UINT64_MAX) {
369 fprintf(stderr, "Исчерпан диапазон времени/ID модели.\n");
370 failed = 1;
371 break;
372 }
373 next_arrival = s.now + interval;
374 }
375 } else {
376 fprintf(stderr, "Нет события, способного продолжить модель.\n");
377 failed = 1;
378 break;
379 }
380 if (!simulation_check(&s)) {
381 fprintf(stderr, "Нарушен инвариант модели на t=%" PRIu64 ".\n",
382 s.now);
383 failed = 1;
384 break;
385 }
386 }
387 uint64_t pending_entrance = 0, pending_painting = 0;
388 for (int i = 0; i < s.slots; ++i) {
389 Visitor *v = &s.visitors[i];
390 if (v->state == ENTRANCE)
391 pending_entrance += s.now - v->wait_since;
392 if (v->state == PAINTING_WAIT)
393 pending_painting += s.now - v->wait_since;
394 }
395 if (s.active)
396 cancel_active(&s);
397 if (!simulation_check(&s)) {
398 fprintf(stderr, "Некорректное освобождение модели.\n");
399 failed = 1;
400 }
401 const char *status = failed || s.io_failed ? "failed"
402 : stop_signal ? "interrupted"
403 : "completed";
404 output_summary(&s, status, pending_entrance, pending_painting);
405 result = failed || s.io_failed ? 1 : stop_signal ? 128 + stop_signal : 0;
406cleanup:
407 if (output_close(&s))
408 result = 1;
409 if (s.visitors)
410 for (int i = 0; i < s.slots; ++i)
411 free(s.visitors[i].seen);
412 free(s.visitors);
413 free(s.membership);
414 free(s.paintings);
415 return result;
416}
