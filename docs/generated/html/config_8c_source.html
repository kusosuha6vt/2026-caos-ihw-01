115static int number(const char *text, uint64_t *value) {
116 if (!text || !isdigit((unsigned char)text[0]))
117 return -1;
118 errno = 0;
119 char *end;
120 uintmax_t parsed = strtoumax(text, &end, 10);
121 if (errno || *end || parsed > UINT64_MAX)
122 return -1;
123 *value = (uint64_t)parsed;
124 return 0;
125}

144static int set_field(Config *c, int field, const char *text) {
145 uint64_t n = 0;
146 if (field == STRATEGY) {
147 for (int i = ORDERED; i <= CYCLIC; ++i)
148 if (!strcmp(text, strategy_name((Strategy)i))) {
149 c->strategy = (Strategy)i;
150 return 0;
151 }
152 } else if (field == LOG || field == JOURNAL) {
153 char *target = field == LOG ? c->log_path : c->journal_path;
154 if (*text && strlen(text) < sizeof(c->log_path)) {
155 strcpy(target, text);
156 return 0;
157 }
158 } else if (!number(text, &n)) {
159 switch (field) {
160 NUMBER_CASE(CAPACITY, capacity)
161 NUMBER_CASE(PAINTINGS, paintings)
162 NUMBER_CASE(PAINTING_CAPACITY, painting_capacity)
163 NUMBER_CASE(VISITORS, visitors)
164 NUMBER_CASE(ARRIVAL_MIN, arrival_min)
165 NUMBER_CASE(ARRIVAL_MAX, arrival_max)
166 NUMBER_CASE(VIEW_MIN, view_min)
167 NUMBER_CASE(VIEW_MAX, view_max)
168 NUMBER_CASE(SEED, seed)
169 NUMBER_CASE(DELAY, delay_ms)
170 NUMBER_CASE(MAX_ACTIVE, max_active)
171 case UNLIMITED:
172 if (n <= 1) {
173 c->unlimited = (int)n;
174 return 0;
175 }
176 break;
177 case QUIET:
178 if (n <= 1) {
179 c->quiet = (int)n;
180 return 0;
181 }
182 break;
183 case JSON:
184 if (n <= 1) {
185 c->json = (int)n;
186 return 0;
187 }
188 break;
189 default:
190 break;
191 }
192 }
193 fprintf(stderr, "Некорректное значение %s: %s\n", keys[field], text);
194 return -1;
195}
196#undef NUMBER_CASE

203static char *trim(char *text) {
204 while (isspace((unsigned char)*text))
205 ++text;
206 char *end = text + strlen(text);
207 while (end > text && isspace((unsigned char)end[-1]))
208 --end;
209 *end = '\0';
210 return text;
211}

220static int read_file(Config *c, const char *path) {
221 int fd = open(path, O_RDONLY);
222 if (fd < 0) {
223 perror(path);
224 return -1;
225 }
226 FILE *input = fdopen(fd, "r");
227 if (!input) {
228 perror("fdopen");
229 close(fd);
230 return -1;
231 }
232 char *line = NULL;
233 size_t allocated = 0;
234 ssize_t length;
235 uint64_t seen = 0;
236 unsigned line_number = 0;
237 int result = 0;
238 while ((length = getline(&line, &allocated, input)) >= 0) {
239 ++line_number;
240 if (length > GALLERY_MAX_INPUT_LINE || (size_t)length != strlen(line)) {
241 result = -1;
242 break;
243 }
244 char *entry = trim(line);
245 if (!*entry || *entry == '#')
246 continue;
247 char *separator = strchr(entry, '=');
248 if (!separator) {
249 result = -1;
250 break;
251 }
252 *separator = '\0';
253 char *key = trim(entry), *value = trim(separator + 1);
254 int field;
255 for (field = 0; field < FIELD_COUNT; ++field)
256 if (!strcmp(key, keys[field]))
257 break;
258 if (field == FIELD_COUNT || (seen & (UINT64_C(1) << field))) {
259 result = -1;
260 break;
261 }
262 seen |= UINT64_C(1) << field;
263 if (set_field(c, field, value)) {
264 result = -1;
265 break;
266 }
267 }
268 if (ferror(input))
269 result = -1;
270 if (result)
271 fprintf(stderr, "Ошибка config %s, строка %u\n", path, line_number);
272 free(line);
273 if (fclose(input))
274 result = -1;
275 return result;
276}

282static int interactive(Config *c) {
283 static const int fields[] = {CAPACITY, PAINTINGS, PAINTING_CAPACITY,
284 VISITORS, ARRIVAL_MIN, ARRIVAL_MAX,
285 VIEW_MIN, VIEW_MAX, STRATEGY};
286 char *line = NULL;
287 size_t size = 0;
288 for (size_t i = 0; i < sizeof(fields) / sizeof(fields[0]); ++i) {
289 fprintf(stderr, "%s: ", keys[fields[i]]);
290 ssize_t length = getline(&line, &size, stdin);
291 /* Match config-file validation: never accept a prefix before NUL. */
292 if (length < 0 || length > GALLERY_MAX_INPUT_LINE ||
293 (size_t)length != strlen(line) ||
294 set_field(c, fields[i], trim(line))) {
295 fprintf(stderr, "Интерактивный ввод прерван или некорректен.\n");
296 free(line);
297 return -1;
298 }
299 }
300 free(line);
301 return 0;
302}

327 if (option == 'h') {
328 help();
329 return 1;
330 }
331 if (option == 'c')
332 file = optarg;
333 if (option == '?') {
334 fprintf(stderr,
335 "Неизвестный/неполный аргумент; используйте --help.\n");

337 }
338 }

341 return -1;
342 }

344 return -1;

346 const char *value = getenv(keys[field]);
347 if (value && set_field(config, field, value))
348 return -1;
349 }
350 optind = 1;
351 while ((option = getopt_long(argc, argv, "", options, NULL)) != -1) {
352 if (option >= 1000 && option < 1000 + FIELD_COUNT) {
353 int field = option - 1000;
354 if (set_field(config, field, field >= UNLIMITED ? "1" : optarg))
355 return -1;
356 } else if (option == 'i')
357 config->interactive = 1;
358 else if (option == 'f')
359 config->unlimited = 0;
360 }
361 if (config->interactive && interactive(config))
362 return -1;
363 if (config->capacity < 1 || config->capacity > 10000 ||
364 config->paintings < 1 || config->paintings > GALLERY_MAX_PAINTINGS ||
365 config->painting_capacity < 1 || config->painting_capacity > 10000 ||
366 config->visitors > 100000 ||
367 config->arrival_min > config->arrival_max ||
368 config->arrival_max > 1000000 || config->view_min < 1 ||
369 config->view_min > config->view_max || config->view_max > 1000000 ||
370 config->delay_ms > 60000 || config->max_active < 1 ||
371 config->max_active > 10000 ||
372 (config->unlimited && config->arrival_max == 0)) {
373 fprintf(stderr,
374 "Параметры вне диапазона или интервалы некорректны; --help.\n");
375 return -1;
376 }
377 if (strcmp(config->log_path, "-") != 0 &&
378 !strcmp(config->log_path, config->journal_path)) {
379 fprintf(stderr, "Журнал и лог должны быть разными файлами.\n");
380 return -1;
381 }
382 return 0;
