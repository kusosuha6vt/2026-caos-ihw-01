23int output_open(Simulation *simulation) {
24 const Config *c = simulation->config;
25 if (strcmp(c->log_path, "-") != 0) {
26 /* Detect existing hard/symbolic aliases before O_TRUNC can damage a
27 * journal. */
28 struct stat log_stat, journal_stat;
29 if (strcmp(c->journal_path, "-") != 0 &&
30 !stat(c->log_path, &log_stat) &&
31 !stat(c->journal_path, &journal_stat) &&
32 log_stat.st_dev == journal_stat.st_dev &&
33 log_stat.st_ino == journal_stat.st_ino) {
34 fprintf(stderr, "Лог и журнал ссылаются на один файл.\n");
35 return -1;
36 }
37 simulation->log_fd =
38 open(c->log_path, O_WRONLY | O_CREAT | O_TRUNC, 0644);
39 if (simulation->log_fd < 0) {
40 perror(c->log_path);
41 return -1;
42 }
43 }
44 return 0;
45}

66static int write_all(int fd, const char *text, size_t length) {
67 while (length) {
68 ssize_t count = write(fd, text, length);
69 if (count < 0 && errno == EINTR) {
70 if (stop_signal)
71 return -1;
72 continue;
73 }
74 if (count <= 0) {
75 if (!count)
76 errno = EIO;
77 return -1;
78 }
79 text += count;
80 length -= (size_t)count;
81 }
82 return 0;
83}

187static int journal(Simulation *s, const char *status, uint64_t pending_entrance,
188 uint64_t pending_painting, double gallery_util,
189 double painting_util) {
190 if (!strcmp(s->config->journal_path, "-"))
191 return 0;
192 int fd = open(s->config->journal_path, O_WRONLY | O_CREAT | O_APPEND, 0644);
193 if (fd < 0)
194 return -1;
195 int result = -1;
196 struct flock lock = {
197 .l_type = F_WRLCK, .l_whence = SEEK_SET, .l_start = 0, .l_len = 0};
198 /* A stuck external journal writer must not make shutdown wait forever. */
199 for (unsigned attempt = 0; fcntl(fd, F_SETLK, &lock) < 0; ++attempt) {
200 if ((errno != EACCES && errno != EAGAIN && errno != EINTR) ||
201 stop_signal || attempt >= 999)
202 goto done;
203 struct timespec pause = {.tv_nsec = 1000000};
204 while (nanosleep(&pause, &pause) < 0)
205 if (errno != EINTR || stop_signal)
206 goto done;
207 }
208 struct stat st;
209 if (fstat(fd, &st))
210 goto done;
211 static const char header[] =
212 "strategy,seed,capacity,paintings,painting_capacity,visitors,"
213 "unlimited,arrival_min,arrival_max,"
214 "view_min,view_max,status,time_ms,arrived,entered,completed,"
215 "cancelled,views_started,views_finished,"
216 "mean_entrance_wait_ms,max_entrance_wait_ms,mean_painting_wait_"
217 "ms,max_painting_wait_ms,"
218 "pending_entrance_wait_ms,pending_painting_wait_ms,peak_"
219 "gallery,gallery_utilization,painting_utilization\n";
220 if (!st.st_size && write_all(fd, header, sizeof(header) - 1))
221 goto done;
222 char row[2048];
223 const Config *c = s->config;
224 snprintf(row, sizeof(row),
225 "%s,%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%" PRIu64
226 ",%d,"
227 "%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%s,%" PRIu64 ","
228 "%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%" PRIu64
229 ",%" PRIu64 ","
230 "%.6f,%" PRIu64 ",%.6f,%" PRIu64 ",%" PRIu64 ",%" PRIu64
231 ",%" PRIu64 ",%.9f,%.9f\n",
232 strategy_name(c->strategy), c->seed, c->capacity, c->paintings,
233 c->painting_capacity, c->visitors, c->unlimited, c->arrival_min,
234 c->arrival_max, c->view_min, c->view_max, status, s->now,
235 s->arrived, s->entered, s->completed, s->cancelled,
236 s->views_started, s->views_finished,
237 s->entered ? (double)s->entrance_wait_sum / (double)s->entered : 0,
238 s->entrance_wait_max,
239 s->views_started
240 ? (double)s->painting_wait_sum / (double)s->views_started
241 : 0,
242 s->painting_wait_max, pending_entrance, pending_painting,
243 s->peak_gallery, gallery_util, painting_util);
244 result = write_all(fd, row, strlen(row));
245done:
246 if (close(fd))
247 result = -1;
248 return result;
249}

335 ".\n"
